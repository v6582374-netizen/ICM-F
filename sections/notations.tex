% === 任务权重与影响参数 ===
\multicolumn{3}{l}{\textbf{任务权重与影响参数}} \\
\midrule
$w_{ij}$ & 任务 $j$ 在职业 $i$ 中的权重；$\sum_j w_{ij}=1$ & $[0,1]$ \\
$w_{ij}(0)$ & 基线任务权重（初始时刻） & $[0,1]$ \\
$w_{ij}(t)$ & 时变任务劳动份额（任务结构） & $[0,1]$ \\
$s_{ij}$ & 任务 $j$ 的替代分数（GenAI减少人力努力的程度） & $\{0.25,0.5,0.75\}$ \\
$c_{ij}$ & 任务 $j$ 的增强分数（GenAI扩展功能需求的程度） & $\{0.25,0.5,0.75\}$ \\
$\alpha_i$ & 职业级替代效应：$\sum_j w_{ij}s_{ij}$ & $[0,1]$ \\
$\beta_i$ & 职业级增强效应：$\sum_j w_{ij}c_{ij}$ & $[0,1]$ \\
$\mu_i$ & 职业净冲击：$\beta_i-\alpha_i$ & $[-1,1]$ \\
$\bar c_i$ & 职业增强潜力：$\sum_j w_{ij}(0) c_{ij}$ & $[0,1]$ \\
\midrule

% === 任务级功能需求模型 ===
\multicolumn{3}{l}{\textbf{任务级功能需求模型}} \\
\midrule
$Q_i(t)$ & 职业 $i$ 的整体功能输出需求指数 & 无量纲 \\
$\ell_{ij}(t)$ & 任务 $j$ 的劳动强度（单位输出的人力努力） & 无量纲 \\
$D_{ij}(t)$ & 任务 $j$ 的人力劳动需求指数 & 无量纲 \\
$\kappa_s$ & 替代速度系数 & $>0$ \\
$\kappa_c$ & 增强对需求的放大因子 & $\ge 0$ \\
$q_{ij}(t)$ & 未归一化任务权重：$w_{ij}[(1-A(t))+A(t)\exp(\eta\mu^{task}_{ij})]$ & $\mathbb{R}_{>0}$ \\
$p_{ij}(t)$ & 任务 $j$ 的相对需求份额：$q_{ij}(t)/\sum_k q_{ik}(t)$ & $[0,1]$ \\
$\eta$ & 冲击强度缩放系数（控制任务结构重排幅度） & $\mathbb{R}_{>0}$ \\
$\tau_{ij}(\epsilon)$ & 任务 $j$ 的有效归零时间：$\min\{t: p_{ij}(t)\le\epsilon\}$ & 年 \\

% === 职业重新定义触发器 ===
\multicolumn{3}{l}{\textbf{职业重新定义触发器}} \\
\midrule
$M_{\text{drop}}(t)$ & 消亡质量（需求降至可忽略的任务的基线权重总和） & $[0,1]$ \\
$\varepsilon$ & "可忽略任务"阈值（如 $0.05$ 表示基线的5\%） & $(0,1)$ \\
$\theta$ & 触发职业重新定义的消亡质量阈值 & $(0,1)$ \\
$\theta_I$ & 排名翻转强度阈值 & $(0,1)$ \\
$\theta_J$ & 换血率阈值 & $(0,1)$ \\
$\delta$ & 任务份额漂移阈值（Jensen--Shannon散度） & $>0$ \\
$t^*$ & 职业重新定义触发时间 & 年 \\
$\mathrm{JS}(\cdot\|\cdot)$ & Jensen--Shannon散度 & $[0,1]$ \\
$\Delta p_{ij}(t_0,t_1)$ & 任务 $j$ 在时间段 $[t_0,t_1]$ 的份额变化 & $[-1,1]$ \\
$r_{ij}(t_0,t_1)$ & 任务 $j$ 对结构变化的贡献度：$\frac{1}{2}|\Delta p_{ij}|$ & $[0,1]$ \\
$\mathrm{rank}_i(j,t)$ & 任务 $j$ 在职业 $i$ 中按 $p_{ij}(t)$ 降序的排名 & $\mathbb{Z}^+$ \\
$\mathrm{Top}_m(p_i(t))$ & 职业 $i$ 按 $p_{ij}(t)$ 降序的前 $m$ 个任务集合 & 集合 \\
$I_i(t_0,t_1)$ & 排名翻转对数（Kendall逆序对数） & $\mathbb{Z}_{\ge 0}$ \\
$\widetilde I_i(t_0,t_1)$ & 归一化排名翻转强度：$2I_i/(N_i(N_i-1))$ & $[0,1]$ \\
$J_i(t_0,t_1;m)$ & Top-$m$ 换血率：$1-|\mathrm{Top}_m(t_0)\cap\mathrm{Top}_m(t_1)|/m$ & $[0,1]$ \\
